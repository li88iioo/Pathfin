cmake_minimum_required(VERSION 3.22)
project(pathfin_ass_jni LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED LIBASS_PREFIX)
    message(FATAL_ERROR "Set LIBASS_PREFIX to this module's pinned static subtitle dependencies")
endif()
foreach(component ass freetype fribidi harfbuzz unibreak)
    add_library(${component} STATIC IMPORTED)
    set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${LIBASS_PREFIX}/lib/lib${component}.a")
endforeach()
add_library(pathfin_ass_jni SHARED src/main/cpp/ass_jni.cpp)
target_include_directories(pathfin_ass_jni PRIVATE vendor/libass)
target_link_libraries(pathfin_ass_jni PRIVATE ass freetype fribidi harfbuzz unibreak jnigraphics m)
target_compile_options(pathfin_ass_jni PRIVATE -Wall -Wextra -Werror=return-type -fvisibility=hidden)
target_link_options(pathfin_ass_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined" "-Wl,--exclude-libs,ALL")
